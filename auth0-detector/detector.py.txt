import math
import os
from datetime import datetime
import requests


AUTH0_DOMAIN = os.getenv("AUTH0_DOMAIN", "your-tenant.us.auth0.com")
AUTH0_CLIENT_ID = os.getenv("AUTH0_CLIENT_ID", "your_client_id")
AUTH0_CLIENT_SECRET = os.getenv("AUTH0_CLIENT_SECRET", "your_client_secret")

TELEGRAM_BOT_TOKEN = os.getenv("TELEGRAM_BOT_TOKEN", "your_bot_token")
TELEGRAM_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID", "your_chat_id")

SPEED_THRESHOLD_KMH = 800


def get_auth0_token():
  url = f"https://{AUTH0_DOMAIN}/oauth/token"
  payload = {
      "client_id": AUTH0_CLIENT_ID,
      "client_secret": AUTH0_CLIENT_SECRET,
      "audience": f"https://{AUTH0_DOMAIN}/api/v2/",
      "grant_type": "client_credentials",
  }
  response = requests.post(url, json=payload, timeout=10)
  response.raise_for_status()
  return response.json()["access_token"]


def get_login_logs(token):
  url = f"https://{AUTH0_DOMAIN}/api/v2/logs"
  headers = {"Authorization": f"Bearer {token}"}
  params = {"q": "type:s", "per_page": 20, "sort": "date:-1"}
  response = requests.get(url, headers=headers, params=params, timeout=10)
  response.raise_for_status()
  return response.json()


def get_ip_location(ip_address):
  url = f"http://ip-api.com/json/{ip_address}"
  response = requests.get(url, timeout=10).json()
  if response.get("status") == "success":
    return {
        "lat": response["lat"],
        "lon": response["lon"],
        "city": response.get("city", "Unknown"),
        "country": response.get("countryCode", "??"),
    }
  return None


def calculate_distance_km(lat1, lon1, lat2, lon2):
  r = 6371.0
  phi1, phi2 = math.radians(lat1), math.radians(lat2)
  dphi = math.radians(lat2 - lat1)
  dlambda = math.radians(lon2 - lon1)
  a = (
      math.sin(dphi / 2) ** 2
      + math.cos(phi1) * math.cos(phi2) * math.sin(dlambda / 2) ** 2
  )
  return 2 * r * math.atan2(math.sqrt(a), math.sqrt(1 - a))


def send_telegram_alert(message):
  url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
  payload = {
      "chat_id": TELEGRAM_CHAT_ID,
      "text": message,
      "parse_mode": "Markdown",
  }
  response = requests.post(url, json=payload, timeout=10)
  response.raise_for_status()


def analyze_logs():
  print("[*] Connecting to Auth0 API...")
  token = get_auth0_token()
  logs = get_login_logs(token)

  user_logins = {}
  for entry in logs:
    user = entry.get("user_name")
    ip = entry.get("ip")
    date_str = entry.get("date")
    if user and ip and date_str:
      user_logins.setdefault(user, []).append({
          "ip": ip,
          "time": datetime.fromisoformat(date_str.replace("Z", "+00:00")),
      })

  for user, events in user_logins.items():
    if len(events) < 2:
      continue

    latest, previous = events[0], events[1]
    if latest["ip"] == previous["ip"]:
      continue

    loc1 = get_ip_location(previous["ip"])
    loc2 = get_ip_location(latest["ip"])
    if not loc1 or not loc2:
      continue

    distance = calculate_distance_km(
        loc1["lat"], loc1["lon"], loc2["lat"], loc2["lon"]
    )
    time_diff_hours = (
        abs((latest["time"] - previous["time"]).total_seconds()) / 3600.0
    )
    time_diff_hours = max(time_diff_hours, 0.001)
    speed = distance / time_diff_hours

    print(
        f"[i] User: {user} | Distance: {distance:.0f} km | Speed:"
        f" {speed:.0f} km/h"
    )

    if speed > SPEED_THRESHOLD_KMH:
      alert_text = (
          "🚨 *SECURITY ALERT: Impossible Travel Detected*\n\n"
          f"👤 *User:* `{user}`\n"
          f"📍 *Previous Login:* {loc1['city']}, {loc1['country']}"
          f" (`{previous['ip']}`)\n"
          f"🌍 *Current Login:* {loc2['city']}, {loc2['country']}"
          f" (`{latest['ip']}`)\n"
          f"📏 *Distance:* {distance:.0f} km in {time_diff_hours * 60:.1f}"
          " min\n"
          f"⚡ *Calculated Speed:* {speed:.0f} km/h\n\n"
          "🛠 *Recommended Action:* Revoke active sessions & verify MFA."
      )
      send_telegram_alert(alert_text)
      print("[!] Alert sent to Telegram!")


if __name__ == "__main__":
  analyze_logs()
